# ============================================================ # CORAL SPECIES AI CLASSIFIER # PYTHON VERSION ALIGNED WITH index.html # # Matches the HTML prediction logic: # Input : 256 x 256 RGB # Normalization : /255.0 # Model : TFLite # Output handling : probabilities OR logits # Temperature : 2.0 for logits # Threshold : 95% # Top-K : 5 # Decision : ACCEPT / REJECTED / UNKNOWN # # IMPORTANT: # This is Python/Colab code that mirrors the prediction logic # of index.html. Python cannot be placed directly inside an # index.html browser page; the web page itself uses JavaScript. # ============================================================ import io import json import time import numpy as np import tensorflow as tf from PIL import Image from google.colab import files from IPython.display import display, clear_output # ============================================================ # SETTINGS - SAME AS index.html # ============================================================ INPUT_HEIGHT = 256 INPUT_WIDTH = 256 CHANNELS = 3 CONFIDENCE_THRESHOLD = 0.95 TEMPERATURE = 2.0 TOP_RESULTS = 5 REJECTION_CLASS = "REJECTED / UNKNOWN" # ============================================================ # 1. UPLOAD IMAGE # ============================================================ clear_output(wait=True) print("Upload coral image, TFLite model, and labels.json") uploaded = files.upload() if not uploaded: raise RuntimeError("No files were uploaded.") # ============================================================ # 2. IDENTIFY UPLOADED FILES # ============================================================ image_file = None model_file = None labels_file = None for filename, content in uploaded.items(): lower_name = filename.lower() if lower_name.endswith( (".jpg", ".jpeg", ".png", ".bmp", ".webp") ): if image_file is None: image_file = filename elif lower_name.endswith(".tflite"): if model_file is None: model_file = filename elif lower_name.endswith(".json"): if labels_file is None: labels_file = filename if image_file is None: raise ValueError("Please upload an image file.") if model_file is None: raise ValueError("Please upload a .tflite model.") if labels_file is None: raise ValueError("Please upload labels.json.") # ============================================================ # 3. LOAD LABELS # Same supported formats as index.html # ============================================================ with open( labels_file, "r", encoding="utf-8" ) as f: data = json.load(f) if isinstance(data, list): labels = data elif ( isinstance(data, dict) and isinstance(data.get("labels"), list) ): labels = data["labels"] elif isinstance(data, dict): try: labels = [ data[key] for key in sorted( data.keys(), key=lambda x: int(x) ) ] except (ValueError, TypeError): labels = list(data.values()) else: raise ValueError( "Unsupported labels.json format." ) if len(labels) == 0: raise ValueError( "labels.json contains no labels." ) # ============================================================ # 4. LOAD TFLITE MODEL # ============================================================ interpreter = tf.lite.Interpreter( model_path=model_file ) interpreter.allocate_tensors() input_details = ( interpreter.get_input_details() ) output_details = ( interpreter.get_output_details() ) # ============================================================ # 5. LOAD AND PREPROCESS IMAGE # Same as index.html # ============================================================ img = Image.open( io.BytesIO( uploaded[image_file] ) ).convert("RGB") img = img.resize( (INPUT_WIDTH, INPUT_HEIGHT), Image.Resampling.BILINEAR ) img_array = np.asarray( img, dtype=np.float32 ) # SAME NORMALIZATION AS index.html img_array = img_array / 255.0 # Shape: 1 x 256 x 256 x 3 img_array = np.expand_dims( img_array, axis=0 ) # ============================================================ # 6. PREPARE TFLITE INPUT # ============================================================ input_info = input_details[0] input_dtype = input_info["dtype"] input_shape = tuple( input_info["shape"] ) if input_dtype == np.float32: model_input = ( img_array.astype(np.float32) ) elif input_dtype in (np.int8, np.uint8): scale, zero_point = ( input_info["quantization"] ) if scale == 0: raise ValueError( "Invalid input quantization scale." ) model_input = ( img_array / scale ) + zero_point if input_dtype == np.uint8: model_input = np.clip( model_input, 0, 255 ) else: model_input = np.clip( model_input, -128, 127 ) model_input = ( model_input.astype(input_dtype) ) else: raise TypeError( f"Unsupported input dtype: {input_dtype}" ) # ============================================================ # 7. TFLITE INFERENCE # Same measurement concept as index.html: # only model inference is timed. # ============================================================ interpreter.set_tensor( input_info["index"], model_input ) inference_start = time.perf_counter() interpreter.invoke() inference_end = time.perf_counter() inference_time_ms = ( inference_end - inference_start ) * 1000.0 # ============================================================ # 8. GET RAW OUTPUT # ============================================================ output_info = output_details[0] raw_output = interpreter.get_tensor( output_info["index"] ) raw_output = np.asarray( raw_output, dtype=np.float32 ).reshape(-1) # ============================================================ # 9. DEQUANTIZE OUTPUT IF NECESSARY # ============================================================ if output_info["dtype"] in ( np.int8, np.uint8 ): scale, zero_point = ( output_info["quantization"] ) if scale != 0: raw_output = ( raw_output - zero_point ) * scale # ============================================================ # 10. CHECK WHETHER OUTPUT IS ALREADY PROBABILITIES # EXACT CONCEPT USED BY index.html # ============================================================ def is_probability_distribution(values): if ( not isinstance(values, np.ndarray) or values.size == 0 ): return False all_in_range = bool( np.all( np.isfinite(values) ) and np.all(values >= 0) and np.all(values <= 1) ) if not all_in_range: return False total = float( np.sum(values) ) return abs( total - 1.0 ) < 0.001 # ============================================================ # 11. TEMPERATURE SOFTMAX # EXACTLY: # logits / temperature -> softmax # ============================================================ def softmax_with_temperature( logits, temperature ): if ( not np.isfinite(temperature) or temperature <= 0 ): raise ValueError( "Temperature must be greater than zero." ) logits = np.asarray( logits, dtype=np.float64 ) if not np.all( np.isfinite(logits) ): raise ValueError( "CNN output contains " "non-finite values." ) scaled = ( logits / temperature ) max_value = np.max( scaled ) exp_values = np.exp( scaled - max_value ) total = np.sum( exp_values ) if ( not np.isfinite(total) or total <= 0 ): raise ValueError( "Unable to calculate Softmax." ) return ( exp_values / total ).astype(np.float32) # ============================================================ # 12. CONVERT OUTPUT TO PROBABILITIES # SAME LOGIC AS index.html # ============================================================ if is_probability_distribution( raw_output ): probabilities = ( raw_output.astype( np.float32 ) ) else: probabilities = ( softmax_with_temperature( raw_output, TEMPERATURE ) ) # ============================================================ # 13. CHECK OUTPUT / LABEL COUNT # ============================================================ if len(probabilities) != len(labels): raise ValueError( "Mismatch between model outputs " "and labels.json.\n" f"Model outputs: {len(probabilities)}\n" f"Labels: {len(labels)}" ) # ============================================================ # 14. FIND TOP PREDICTION # ============================================================ predicted_index = int( np.argmax( probabilities ) ) cnn_confidence = float( probabilities[ predicted_index ] ) cnn_predicted_class = str( labels[ predicted_index ] ) # ============================================================ # 15. ACCEPT / REJECT # SAME AS index.html # ============================================================ if ( cnn_confidence >= CONFIDENCE_THRESHOLD ): decision = "ACCEPT" final_prediction = ( cnn_predicted_class ) else: decision = "REJECT" final_prediction = ( REJECTION_CLASS ) # ============================================================ # 16. TOP-5 # ============================================================ sorted_indices = np.argsort( probabilities )[::-1] top_k = min( TOP_RESULTS, len(probabilities) ) top5 = [] for rank in range(top_k): index = int( sorted_indices[rank] ) score = float( probabilities[index] ) top5.append( ( rank + 1, str(labels[index]), score ) ) # ============================================================ # 17. DISPLAY RESULT # ============================================================ print() print("=" * 60) print("CORAL SPECIES CLASSIFICATION AI") print("=" * 60) print( f"\nPredicted Class : " f"{cnn_predicted_class}" ) print( f"CNN Confidence : " f"{cnn_confidence * 100:.2f}%" ) print( f"Threshold : " f"{CONFIDENCE_THRESHOLD * 100:.2f}%" ) print( f"Decision : " f"{decision}" ) print( f"Final Prediction : " f"{final_prediction}" ) print( f"Inference Time : " f"{inference_time_ms:.2f} ms" ) print() print("-" * 60) print("TOP-5 PREDICTIONS") print("-" * 60) for rank, name, score in top5: print( f"{rank}. " f"{name:<30} " f"{score * 100:>7.2f}%" ) print() print("=" * 60) if decision == "ACCEPT": print("✓ ACCEPTED") print( f"Class: {cnn_predicted_class}" ) else: print("✗ REJECTED / UNKNOWN") print( f"Top CNN Class: " f"{cnn_predicted_class}" ) print( f"Reason: confidence " f"is below " f"{CONFIDENCE_THRESHOLD * 100:.0f}%." ) print("=" * 60)
